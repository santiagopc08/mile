'use client';
import { memo } from 'react';

import { useState, useEffect } from 'react';
import { motion, AnimatePresence } from 'framer-motion';
import { Trash2, Check, Pencil, Play, RotateCcw, Ban, ChevronRight, CheckSquare, Square } from 'lucide-react';
import { FuturisticProgressBar } from '@/components/ui/FuturisticProgressBar';
import { ChamferedPanel } from '@/components/ui/ChamferedPanel';
import { LiveLinkPreview } from '@/components/LiveLinkPreview';
import { Task, Objective, categoryStyles, priorityStyles } from './taskTypes';

interface TaskCardProps {
  task: Task;
  isEditing: boolean;
  onEditStart: (task: Task) => void;
  onEditCancel: () => void;
  onEditSave: (updatedTask: Task) => void;
  onDelete: (id: string) => void;
  onUpdateStatus: (id: string, status: Task['status']) => void;
  onToggleChecklist: (taskId: string, listType: 'actions' | 'validations', itemId: string) => void;
  expandedTaskId: string | null;
  setExpandedTaskId: (id: string | null) => void;
  getTaskObjective: (task: Task) => Objective | undefined;
  accentClass: string;
  accentColor: string;
  profile: 'el' | 'ella' | null;
  isLateOrOverflow: boolean;
}


// ⚡ Bolt Optimization: Wrap component in React.memo()
// Expected Impact: Prevents expensive re-renders of list items during parent state updates (e.g. active item changes), saving ~3-5ms per render cycle in long lists.
export const TaskCard = memo(function TaskCard({
  task,
  isEditing,
  onEditStart,
  onEditCancel,
  onEditSave,
  onDelete,
  onUpdateStatus,
  onToggleChecklist,
  expandedTaskId,
  setExpandedTaskId,
  getTaskObjective,
  accentClass,
  accentColor,
  profile,
  isLateOrOverflow,
}: TaskCardProps) {
  const taskPriority = task.priority || 'medium';
  const categoryStyle = categoryStyles[task.category];
  const isCollapsed = task.status === 'done' || task.status === 'skipped';

  const taskAuthor = task.assignee || profile || 'el';
  const authorColor = taskAuthor === 'ella' ? '#ff4b89' : '#c3f400';
  const authorName = taskAuthor === 'ella' ? 'MILENA' : 'SANTIAGO';

  let checkedActionsCount = 0;
  if (!isCollapsed && task.actions) {
    for (const a of task.actions) if (a.checked) checkedActionsCount++;
  }

  let checkedValidationsCount = 0;
  if (!isCollapsed && task.validations) {
    for (const v of task.validations) if (v.checked) checkedValidationsCount++;
  }

  // Edit state
  const [editTaskText, setEditTaskText] = useState(task.text);
  const [editTaskCategory, setEditTaskCategory] = useState(task.category);
  const [editTaskObjectiveId, setEditTaskObjectiveId] = useState(task.objective_id || '');
  const [editEstimatedTime, setEditEstimatedTime] = useState(task.estimated_time);
  const [editActualTime, setEditActualTime] = useState(task.actual_time);
  const [editDueDate, setEditDueDate] = useState(task.due_date || '');
  const [editPriority, setEditPriority] = useState(task.priority || 'medium');
  const [editActions, setEditActions] = useState<NonNullable<Task['actions']>>(task.actions || []);
  const [editValidations, setEditValidations] = useState<NonNullable<Task['actions']>>(task.validations || []);
  const [editDetail, setEditDetail] = useState(task.detail || '');
  const [editTaskAssignee, setEditTaskAssignee] = useState(task.assignee || profile || 'el');

  // Reset edit state when editing starts for this task
  useEffect(() => {
    if (isEditing) {
      setEditTaskText(task.text);
      setEditTaskCategory(task.category);
      setEditTaskObjectiveId(task.objective_id || '');
      setEditEstimatedTime(task.estimated_time);
      setEditActualTime(task.actual_time);
      setEditDueDate(task.due_date || '');
      setEditPriority(task.priority || 'medium');
      setEditActions(task.actions || []);
      setEditValidations(task.validations || []);
      setEditDetail(task.detail || '');
      setEditTaskAssignee(task.assignee || profile || 'el');
    }
  }, [isEditing, task, profile]);

  const handleSave = () => {
    onEditSave({
      ...task,
      text: editTaskText,
      category: editTaskCategory,
      objective_id: editTaskObjectiveId || undefined,
      estimated_time: editEstimatedTime,
      actual_time: editActualTime,
      due_date: editDueDate || undefined,
      priority: editPriority,
      actions: editActions,
      validations: editValidations,
      detail: editDetail.trim() || undefined,
      assignee: editTaskAssignee,
    });
  };

  const objective = getTaskObjective(task);
  const isExpanded = expandedTaskId === task.id;
  const progress = task.estimated_time > 0 ? Math.min((task.actual_time / task.estimated_time) * 100, 100) : 0;

  // Header line: Name on left, Category & Priority chips together on upper right
  const headerLabel = authorName;

  const headerRightElement = (
    <div className="flex items-center gap-1.5">
      {/* Category Pill */}
      <span
        className={`text-[7px] font-mono uppercase font-bold tracking-wider px-1.5 py-0.5 border border-white/10 ${categoryStyle.chip}`}
      >
        {categoryStyle.label}
      </span>

      {/* Priority Pill */}
      <span
        className={`text-[7px] font-mono uppercase font-bold tracking-wider px-1.5 py-0.5 border border-white/10 ${priorityStyles[taskPriority].chip}`}
      >
        {priorityStyles[taskPriority].label}
      </span>
    </div>
  );

  return (
    <motion.div
      layout
      initial={{ opacity: 0, y: 8 }}
      animate={{ opacity: 1, y: 0 }}
      exit={{ opacity: 0, scale: 0.96 }}
      className="group relative w-full"
    >
      <ChamferedPanel
        accentColor={authorColor}
        notchSize={14}
        label={headerLabel}
        headerRight={headerRightElement}
        showSideTabs={false}
        className={`!p-3.5 flex flex-col justify-between transition-all duration-300 ${
          isCollapsed ? 'opacity-40 hover:opacity-80' : ''
        } ${isEditing ? '!border-white shadow-[0_0_15px_rgba(255,255,255,0.15)]' : ''}`}
      >
        {isEditing ? (
          <div className="space-y-3">
            <div className="flex items-center justify-between border-b border-white/10 pb-2">
              <span className="text-[8.5px] font-mono uppercase tracking-widest text-white font-bold">
                EDITAR TAREA
              </span>
              <div className="flex gap-2">
                <button
                  type="button"
                  onClick={onEditCancel}
                  className="px-2.5 py-1 text-[8px] font-mono border border-white/20 text-stone-400 hover:text-white transition-colors"
                >
                  CANCELAR
                </button>
                <button
                  type="button"
                  onClick={handleSave}
                  className="px-3 py-1 text-[8px] font-mono font-bold border text-black transition-opacity hover:opacity-90"
                  style={{ backgroundColor: authorColor, borderColor: authorColor }}
                >
                  GUARDAR
                </button>
              </div>
            </div>

            <div>
              <label className="text-[7px] uppercase font-bold text-stone-400 font-mono block mb-1">Título</label>
              <input
                value={editTaskText}
                onChange={e => setEditTaskText(e.target.value)}
                className="w-full bg-black border border-white/20 px-2.5 py-2 text-[11px] font-mono text-white outline-none focus:border-white"
              />
            </div>

            <div className="grid grid-cols-2 gap-2">
              <div>
                <label className="text-[7px] uppercase font-bold text-stone-400 font-mono block mb-1">Categoría</label>
                <select
                  value={editTaskCategory}
                  onChange={e => setEditTaskCategory(e.target.value as 'work' | 'home' | 'personal')}
                  className="w-full h-[34px] bg-black border border-white/20 px-2 text-[9px] uppercase text-stone-300 outline-none font-mono"
                >
                  <option value="work">TRABAJO</option>
                  <option value="home">HOGAR</option>
                  <option value="personal">PERSONAL</option>
                </select>
              </div>
              <div>
                <label className="text-[7px] uppercase font-bold text-stone-400 font-mono block mb-1">Prioridad</label>
                <select
                  value={editPriority}
                  onChange={e => setEditPriority(e.target.value as 'low' | 'medium' | 'high')}
                  className="w-full h-[34px] bg-black border border-white/20 px-2 text-[9px] uppercase text-stone-300 outline-none font-mono"
                >
                  <option value="low">BAJA</option>
                  <option value="medium">MEDIA</option>
                  <option value="high">ALTA</option>
                </select>
              </div>
            </div>

            <div className="grid grid-cols-2 gap-2">
              <div>
                <label className="text-[7px] uppercase font-bold text-stone-400 font-mono block mb-1">Est. (Minutos)</label>
                <input
                  type="number"
                  value={editEstimatedTime}
                  onChange={e => setEditEstimatedTime(Number(e.target.value))}
                  className="w-full h-[34px] bg-black border border-white/20 px-2 text-[10px] font-mono text-white outline-none"
                />
              </div>
              <div>
                <label className="text-[7px] uppercase font-bold text-stone-400 font-mono block mb-1">Real (Minutos)</label>
                <input
                  type="number"
                  value={editActualTime}
                  onChange={e => setEditActualTime(Number(e.target.value))}
                  className="w-full h-[34px] bg-black border border-white/20 px-2 text-[10px] font-mono text-white outline-none"
                />
              </div>
            </div>

            <div>
              <label className="text-[7px] uppercase font-bold text-stone-400 font-mono block mb-1">Detalle / Notas</label>
              <textarea
                value={editDetail}
                onChange={e => setEditDetail(e.target.value)}
                rows={2}
                className="w-full bg-black border border-white/20 px-2.5 py-1.5 text-[9.5px] font-mono text-stone-300 outline-none resize-none"
              />
              {editDetail.match(/https?:\/\/[^\s]+/) && (
                <div className="pt-1">
                  <LiveLinkPreview url={editDetail.match(/https?:\/\/[^\s]+/)![0]} label="ENLACE DETECTADO" />
                </div>
              )}
            </div>
          </div>
        ) : (
          <>
            {/* Main Content Body */}
            <div className="w-full min-w-0 flex flex-col gap-1.5 justify-center py-1">
              {/* Objective Tag if assigned */}
              {objective && (
                <div className="mb-0.5">
                  <span className="inline-flex items-center gap-1 text-[7.5px] font-mono uppercase tracking-wider text-stone-400 border border-white/10 bg-white/[0.02] px-1.5 py-0.5 max-w-full truncate">
                    <span>🎯</span>
                    <span className="truncate">{objective.title}</span>
                  </span>
                </div>
              )}

              <h4
                className={`text-xs sm:text-sm font-black uppercase tracking-wider font-mono leading-snug ${
                  isCollapsed ? 'line-through text-[#a88a7e]' : 'text-white'
                }`}
              >
                {task.text}
              </h4>

              {task.detail && (
                <p className="text-[10px] leading-relaxed text-[#a88a7e] line-clamp-2 font-sans">
                  {task.detail}
                </p>
              )}

              {/* Single Time & Progress Bar */}
              {!isCollapsed && task.estimated_time > 0 && (
                <div className="mt-1.5">
                  <div className="flex items-center justify-between mb-1">
                    <span className="text-[9.5px] font-mono font-bold" style={{ color: authorColor }}>
                      {task.actual_time} min
                    </span>
                    <span className="text-[9.5px] font-mono text-white/40">
                      / {task.estimated_time} min · {Math.round(progress)}%
                    </span>
                  </div>
                  <FuturisticProgressBar
                    progress={progress}
                    color={authorColor}
                  />
                </div>
              )}

              {isCollapsed && task.estimated_time > 0 && (
                <div className="mt-1 text-[8.5px] font-mono font-bold text-emerald-400">
                  ✓ {task.actual_time} min dedicados
                </div>
              )}

              {/* Action / Validation Sub-checklists Toggles */}
              {!isCollapsed && ((task.actions && task.actions.length > 0) || (task.validations && task.validations.length > 0)) && (
                <div className="flex items-center gap-2 mt-1.5 pt-1 border-t border-white/5">
                  {task.actions && task.actions.length > 0 && (
                    <button
                      type="button"
                      onClick={() => setExpandedTaskId(isExpanded ? null : task.id)}
                      className={`text-[7.5px] font-mono uppercase font-bold tracking-wider px-2 py-0.5 border transition-colors flex items-center gap-1 ${
                        isExpanded
                          ? 'border-white bg-white/10 text-white'
                          : 'border-white/10 bg-black/40 text-stone-400 hover:text-white'
                      }`}
                    >
                      <CheckSquare size={9} />
                      <span>PASOS {checkedActionsCount}/{task.actions.length}</span>
                    </button>
                  )}

                  {task.validations && task.validations.length > 0 && (
                    <button
                      type="button"
                      onClick={() => setExpandedTaskId(isExpanded ? null : task.id)}
                      className={`text-[7.5px] font-mono uppercase font-bold tracking-wider px-2 py-0.5 border transition-colors flex items-center gap-1 ${
                        isExpanded
                          ? 'border-white bg-white/10 text-white'
                          : 'border-white/10 bg-black/40 text-stone-400 hover:text-white'
                      }`}
                    >
                      <span>VLD {checkedValidationsCount}/{task.validations.length}</span>
                    </button>
                  )}
                </div>
              )}

              {/* Collapsible Drawer for Checklist Items */}
              <AnimatePresence>
                {!isCollapsed && isExpanded && (
                  <motion.div
                    initial={{ height: 0, opacity: 0 }}
                    animate={{ height: 'auto', opacity: 1 }}
                    exit={{ height: 0, opacity: 0 }}
                    className="overflow-hidden space-y-2 border-t border-white/10 pt-2 mt-1 bg-black/60 p-2 border border-white/5"
                  >
                    {task.actions && task.actions.length > 0 && (
                      <div className="space-y-1">
                        <span className="text-[7px] text-stone-400 font-mono tracking-widest uppercase block mb-1">
                          Pasos de Acción
                        </span>
                        {task.actions.map(act => (
                          <button
                            key={act.id}
                            type="button"
                            onClick={() => onToggleChecklist(task.id, 'actions', act.id)}
                            className={`w-full min-h-[28px] px-2 py-1 flex items-center gap-2 text-[9.5px] text-left font-mono border transition-colors ${
                              act.checked
                                ? 'border-white/5 bg-white/[0.01] text-stone-500 line-through'
                                : 'border-white/10 bg-white/[0.03] text-stone-200 hover:border-white/25 hover:text-white'
                            }`}
                          >
                            {act.checked ? (
                              <CheckSquare size={12} className="text-emerald-400 shrink-0" />
                            ) : (
                              <Square size={12} className="text-stone-500 shrink-0" />
                            )}
                            <span className="truncate">{act.text}</span>
                          </button>
                        ))}
                      </div>
                    )}

                    {task.validations && task.validations.length > 0 && (
                      <div className="space-y-1">
                        <span className="text-[7px] text-stone-400 font-mono tracking-widest uppercase block mb-1">
                          Criterios de Validación
                        </span>
                        {task.validations.map(val => (
                          <button
                            key={val.id}
                            type="button"
                            onClick={() => onToggleChecklist(task.id, 'validations', val.id)}
                            className={`w-full min-h-[28px] px-2 py-1 flex items-center gap-2 text-[9.5px] text-left font-mono border transition-colors ${
                              val.checked
                                ? 'border-white/5 bg-white/[0.01] text-stone-500 line-through'
                                : 'border-white/10 bg-white/[0.03] text-stone-200 hover:border-white/25 hover:text-white'
                            }`}
                          >
                            {val.checked ? (
                              <CheckSquare size={12} className="text-emerald-400 shrink-0" />
                            ) : (
                              <Square size={12} className="text-stone-500 shrink-0" />
                            )}
                            <span className="truncate">{val.text}</span>
                          </button>
                        ))}
                      </div>
                    )}
                  </motion.div>
                )}
              </AnimatePresence>
            </div>

            {/* Footer Controls Row */}
            <div className="flex items-stretch border-t border-white/10 mt-2.5 pt-2">
              {task.status === 'todo' && (
                <button
                  type="button"
                  onClick={() => onUpdateStatus(task.id, 'in_progress')}
                  className="flex flex-1 items-center justify-center gap-1.5 py-1.5 text-[8.5px] font-mono font-bold uppercase tracking-widest transition-colors text-[#00dbe9] hover:bg-white/5"
                >
                  <Play size={10} className="fill-current" />
                  <span>INICIAR TAREA</span>
                  <ChevronRight className="h-3 w-3" />
                </button>
              )}

              {task.status === 'in_progress' && (
                <>
                  <button
                    type="button"
                    onClick={() => onUpdateStatus(task.id, 'skipped')}
                    className="flex flex-1 items-center justify-center gap-1 py-1.5 text-[8px] font-mono font-bold uppercase tracking-widest transition-colors text-stone-400 hover:text-rose-400 hover:bg-white/5 border-r border-white/10"
                    title="Omitir tarea"
                  >
                    <Ban size={10} />
                    <span>OMITIR</span>
                  </button>
                  <button
                    type="button"
                    onClick={() => onUpdateStatus(task.id, 'done')}
                    className="flex flex-1 items-center justify-center gap-1.5 py-1.5 text-[8.5px] font-mono font-bold uppercase tracking-widest transition-colors text-emerald-400 hover:bg-white/5"
                  >
                    <Check size={12} strokeWidth={2.5} />
                    <span>COMPLETAR</span>
                    <ChevronRight className="h-3 w-3" />
                  </button>
                </>
              )}

              {(task.status === 'done' || task.status === 'skipped') && (
                <button
                  type="button"
                  onClick={() => onUpdateStatus(task.id, 'in_progress')}
                  className="flex flex-1 items-center justify-center gap-1.5 py-1.5 text-[8px] font-mono font-bold uppercase tracking-widest transition-colors text-stone-300 hover:text-white hover:bg-white/5"
                >
                  <RotateCcw size={10} />
                  <span>RETOMAR TAREA</span>
                </button>
              )}

              {/* Edit Icon Button */}
              {!isCollapsed && (
                <button
                  type="button"
                  onClick={() => onEditStart(task)}
                  className="flex w-9 items-center justify-center border-l border-white/10 text-stone-400 transition-colors hover:text-[#00dbe9]"
                  title="Editar tarea"
                >
                  <Pencil className="w-3.5 h-3.5" />
                </button>
              )}

              {/* Delete Icon Button */}
              <button
                type="button"
                onClick={() => onDelete(task.id)}
                className="flex w-9 items-center justify-center border-l border-white/10 text-stone-400 transition-colors hover:text-rose-400"
                title="Eliminar tarea"
              >
                <Trash2 className="w-3.5 h-3.5" />
              </button>
            </div>
          </>
        )}
      </ChamferedPanel>
    </motion.div>
  );
});
